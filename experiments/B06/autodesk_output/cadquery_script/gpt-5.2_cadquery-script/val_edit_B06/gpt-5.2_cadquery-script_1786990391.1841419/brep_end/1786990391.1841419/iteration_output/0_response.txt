    def _safe_dot(a, b):
        return a.x * b.x + a.y * b.y + a.z * b.z

    def _get_face_normal(face):
        # Try to compute normal at face center robustly
        pt = face.Center()
        try:
            u, v = face.paramAt(pt)
            return face.normalAt(u, v)
        except Exception:
            try:
                return face.normalAt()
            except Exception:
                # Fallback: approximate with a planar normal if possible
                try:
                    return face.normalAt(0, 0)
                except Exception:
                    return cq.Vector(0, 0, 1)

    def _edge_is_closed(edge):
        try:
            return bool(edge.isClosed())
        except Exception:
            try:
                return bool(edge.wrapped.Closed())
            except Exception:
                return False

    def chamfer_hole_edges_on_solid(solid, chamfer_size=0.2):
        bbox = solid.BoundingBox()
        c = bbox.center
        internal_cyl_faces = []
        hole_edges = []

        faces = solid.Faces()
        for f in faces:
            try:
                if str(f.geomType()).upper() != "CYLINDER":
                    continue
            except Exception:
                continue

            n = _get_face_normal(f)
            fc = f.Center()
            v = cq.Vector(fc.x - c.x, fc.y - c.y, fc.z - c.z)
            d = _safe_dot(n, v)

            # Heuristic:

            # For hole interior cylindrical faces, outward normal points toward the part center => dot < 0
            if d < 0:

        # Collect closed circular edges from internal cylindrical faces
        wrapped_seen = set()
        for f in internal_cyl_faces:

                if not _edge_is_closed(e):
