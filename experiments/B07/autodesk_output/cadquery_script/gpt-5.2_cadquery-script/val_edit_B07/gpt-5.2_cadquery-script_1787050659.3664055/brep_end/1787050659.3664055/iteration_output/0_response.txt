    def _edge_dir_is_y(edge, tol=0.90):
        vs = edge.Vertices()
        if len(vs) < 2:
            return False
        p1 = cq.Vector(*vs[0].toTuple())
        p2 = cq.Vector(*vs[-1].toTuple())
        d = p2 - p1
        n = d.Length
        if n < 1e-9:
            return False
        d = d / n
        return abs(d.y) > tol

    def _collect_vertical_line_edges(s):
        out = []

                continue
            if not _edge_dir_is_y(e):

            out.append((e, r, c.y, e.Length()))
