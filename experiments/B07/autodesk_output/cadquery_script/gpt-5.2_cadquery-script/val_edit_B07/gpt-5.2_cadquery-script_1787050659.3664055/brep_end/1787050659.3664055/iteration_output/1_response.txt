def my_cad_function(args):
    import cadquery as cq
    import os, math

    if "input_file" not in args:
        print("No input_file provided")
        return None

    step_path = os.path.expanduser(args["input_file"])
    wp = cq.importers.importStep(step_path)
    shape = wp.val() if hasattr(wp, "val") else wp

    bb = shape.BoundingBox()
    center = bb.center
    max_dim = max(bb.xlen, bb.ylen, bb.zlen)
    print(f"Loaded STEP: {step_path}")
    print(f"BBox lens: x={bb.xlen:.6f}, y={bb.ylen:.6f}, z={bb.zlen:.6f}, max_dim={max_dim:.6f}")
    print(f"Faces: {len(shape.Faces())}, Edges: {len(shape.Edges())}")

    # Fusion 360 API/internal units are centimeters; STEP exported from Fusion commonly ends up in cm.
    # Heuristic: if overall part size is < 50 units, treat as cm.
    if max_dim < 50:
        unit = "cm"
        mm_to_u = 0.1
    else:
        unit = "mm"
        mm_to_u = 1.0

    r_outer = 2.0 * mm_to_u
    r_inner = 1.0 * mm_to_u
    print(f"Assumed units: {unit}  (mm_to_u={mm_to_u})")
    print(f"Target fillets: outer={r_outer} u (2mm), inner={r_inner} u (1mm)")

    def edge_dir(edge):
        vs = edge.Vertices()
        if len(vs) < 2:
            return None
        p1 = cq.Vector(*vs[0].toTuple())
        p2 = cq.Vector(*vs[-1].toTuple())
        d = p2 - p1
        L = d.Length
        if L < 1e-9:
            return None
        return d / L

    def collect_vertical_line_edges(s, y_axis_bias=0.75):
        infos = []
        for e in s.Edges():
            try:
                if e.geomType() != "LINE":
                    continue
            except Exception:
                continue
            d = edge_dir(e)
            if d is None:
                continue
            if abs(d.y) < y_axis_bias:
                continue
            c = e.Center()
            rx = c.x - center.x
            rz = c.z - center.z
            r = math.hypot(rx, rz)
            infos.append((e, r, e.Length(), c.y))
        return infos

    def bin_by_radius(edge_infos, bin_size):
        bins = {}
        for (e, r, L, cy) in edge_infos:
            k = round(r / bin_size) * bin_size
            bins.setdefault(k, []).append((e, r, L, cy))
        # compute mean radius for each bin
        out = []
        for k, v in bins.items():
            mr = sum(t[1] for t in v) / len(v)
            out.append((k, mr, v))
        return bins, sorted(out, key=lambda t: t[1])

    def pick_inner_outer_bins(edge_infos):
        # Bin size relative to part size
        bin_size = max(1e-6, 0.01 * max_dim)
        _, binfo = bin_by_radius(edge_infos, bin_size)

        # Candidate bins: exactly 4 edges, and not too close to center or too far out
        candidates = []
        for (k, mr, v) in binfo:
            if len(v) != 4:
                continue
            if mr < 0.05 * max_dim:
                continue
            if mr > 0.70 * max_dim:
                continue
            candidates.append((k, mr, v))

        print(f"Vertical LINE edges: {len(edge_infos)}")
        print(f"4-edge radius bins found: {[round(c[1],6) for c in candidates]}")

        if len(candidates) >= 2:
            # smallest-radius bin -> inner, next -> outer (for the central square/frustum feature)
            inner = candidates[0]
            outer = candidates[1]
            return inner, outer
        elif len(candidates) == 1:
            # If only one 4-edge bin, treat it as inner (more likely), outer will be approximated later
            return candidates[0], None
        else:
            return None, None

    # --- Select target edge sets on the original shape ---
    vlines0 = collect_vertical_line_edges(shape, y_axis_bias=0.70)
    inner0, outer0 = pick_inner_outer_bins(vlines0)

    outer_target_r = None
    inner_edges0 = []
    outer_edges0 = []

    if inner0 is not None:
        inner_target_r = inner0[1]
        inner_edges0 = [t[0] for t in inner0[2]]
        print(f"Picked INNER edges: r~{inner_target_r:.6f}, count={len(inner_edges0)}")
    if outer0 is not None:
        outer_target_r = outer0[1]
        outer_edges0 = [t[0] for t in outer0[2]]
        print(f"Picked OUTER edges: r~{outer_target_r:.6f}, count={len(outer_edges0)}")

    # Fallback if bins did not resolve as expected: pick 4 edges with two radii clusters
    if not outer_edges0 and len(vlines0) >= 8:
        # sort by radius and take two quartets around lower-middle and upper-middle radii
        v_sorted = sorted(vlines0, key=lambda t: t[1])
        q1 = v_sorted[max(0, len(v_sorted)//3 - 2): max(0, len(v_sorted)//3 - 2) + 4]
        q2 = v_sorted[max(0, 2*len(v_sorted)//3 - 2): max(0, 2*len(v_sorted)//3 - 2) + 4]
        # decide which is inner/outer by mean radius
        mr1 = sum(t[1] for t in q1) / 4
        mr2 = sum(t[1] for t in q2) / 4
        if mr1 < mr2:
            if not inner_edges0:
                inner_edges0 = [t[0] for t in q1]
                inner_target_r = mr1
            outer_edges0 = [t[0] for t in q2]
            outer_target_r = mr2
        else:
            if not inner_edges0:
                inner_edges0 = [t[0] for t in q2]
                inner_target_r = mr2
            outer_edges0 = [t[0] for t in q1]
            outer_target_r = mr1
        print("Fallback clustering used for INNER/OUTER selection")
        print(f"INNER r~{inner_target_r:.6f} count={len(inner_edges0)}")
        print(f"OUTER r~{outer_target_r:.6f} count={len(outer_edges0)}")

    # --- Apply outer fillet (2mm) ---
    shape1 = shape
    if outer_edges0:
        try:
            shape1 = shape1.fillet(r_outer, outer_edges0)
            print(f"Applied OUTER fillet (2mm) on {len(outer_edges0)} edges")
        except Exception as e:
            print(f"OUTER fillet failed: {e}")

    # --- Re-select inner edges on the modified shape and apply inner fillet (1mm) ---
    shape2 = shape1
    vlines1 = collect_vertical_line_edges(shape1, y_axis_bias=0.70)

    inner_edges1 = []
    if vlines1:
        bin_size = max(1e-6, 0.01 * max_dim)
        bins, binfo = bin_by_radius(vlines1, bin_size)
        keys4 = [(k, mr, v) for (k, mr, v) in binfo if len(v) == 4]

        chosen = None
        if inner_target_r is not None and keys4:
            chosen = min(keys4, key=lambda t: abs(t[1] - inner_target_r))
        elif keys4:
            chosen = keys4[0]

        if chosen is not None:
            inner_edges1 = [t[0] for t in chosen[2]]
            print(f"Re-picked INNER edges: r~{chosen[1]:.6f}, count={len(inner_edges1)}")

    if inner_edges1:
        try:
            shape2 = shape2.fillet(r_inner, inner_edges1)
            print(f"Applied INNER fillet (1mm) on {len(inner_edges1)} edges")
        except Exception as e:
            print(f"INNER fillet failed: {e}")

    return shape2
