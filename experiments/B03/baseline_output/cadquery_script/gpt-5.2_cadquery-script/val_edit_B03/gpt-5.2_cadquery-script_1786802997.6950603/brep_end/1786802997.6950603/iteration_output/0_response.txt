def my_cad_function(args):
    import os
    import cadquery as cq

    # --- Load STEP ---
    step_path = args.get("input_file", "")
    step_path = os.path.expanduser(step_path) if step_path else step_path
    if not step_path or (not os.path.exists(step_path)):
        print("No valid input_file provided in args. args keys:", list(args.keys()))
        return None

    model_wp = cq.importers.importStep(step_path)
    occ = model_wp.val() if hasattr(model_wp, "val") else model_wp

    # If compound/multi-solid, pick the largest solid for modification
    try:
        solids = occ.Solids() if hasattr(occ, "Solids") else []
    except Exception:
        solids = []

    if solids:
        main = max(solids, key=lambda s: s.Volume())
    else:
        main = occ

    bbox = main.BoundingBox()
    max_dim = max(bbox.xlen, bbox.ylen, bbox.zlen)

    # Heuristic: if model is in cm-like scale (wheel bbox ~40-60), use 0.1 as 1mm
    chamfer_size = 1.0 if max_dim > 100 else 0.1

    print(f"Loaded: {os.path.basename(step_path)}")
    print(f"BBox lens: x={bbox.xlen:.3f}, y={bbox.ylen:.3f}, z={bbox.zlen:.3f}, max_dim={max_dim:.3f}")
    print(f"Chamfer size chosen (1mm): {chamfer_size}")

    # --- Find the valve hole circular edges (both sides) ---
    edges = list(main.Edges())
    center_bb = bbox.center

    # Filter to small circular edges (avoid rim main circles)
    rmin = 2.0 * chamfer_size
    rmax = 30.0 * chamfer_size

    circ = []
    for e in edges:
        try:
            if e.geomType() != "CIRCLE":
                continue
            r = e.radius()
            if not (rmin <= r <= rmax):
                continue
            c = e.Center()
            d = (c.sub(center_bb)).Length
            circ.append((e, r, c, d))
        except Exception:
            continue

    print(f"Circular edge candidates (small radii): {len(circ)}")
    if not circ:
        print("No suitable circular edges found; returning original model")
        return cq.Workplane(obj=main)

    # Pick the circular edge farthest from the bbox center (likely the valve hole at outer rim)
    e0, r0, c0, d0 = max(circ, key=lambda t: t[3])

    tol_c = 1.5 * chamfer_size  # center coincidence tolerance
    tol_r = 0.75 * chamfer_size

    selected = []
    for e, r, c, d in circ:
        if (c.sub(c0)).Length <= tol_c and abs(r - r0) <= tol_r:
            selected.append(e)

    print(f"Seed edge radius={r0:.4f}, center=({c0.x:.4f},{c0.y:.4f},{c0.z:.4f}), dist_from_bbox_center={d0:.4f}")
    print(f"Selected circular edges for chamfer: {len(selected)}")

    if len(selected) < 2:
        # Still proceed; sometimes only one opening edge exists depending on topology
        print("Warning: fewer than 2 edges selected; proceeding anyway")

    base = cq.Workplane(obj=main)
    try:
        result = base.newObject(selected).chamfer(chamfer_size)
        return result
    except Exception as e:
        print("Chamfer failed:", e)
        import traceback
        traceback.print_exc()
        return cq.Workplane(obj=main)
