def my_cad_function(args):
    import os, math
    import cadquery as cq

    # Load the provided STEP
    step_fallback = r"C:/PROGRAMING_PYTHON/2026-08-14_ASME_Hackathon_2026/neuralCAD-Edit-data/edit_192_external\\breps/B7A2N74ZJBF9MZHU_1770156951.5787919.step"
    step_path = os.path.expanduser(args.get("input_file", step_fallback))
    model = cq.importers.importStep(step_path)

    shape = model.val() if hasattr(model, "val") else model
    bbox = shape.BoundingBox()
    c0 = bbox.center

    # Infer wheel axis as the smallest bbox dimension
    dims = {"X": bbox.xlen, "Y": bbox.ylen, "Z": bbox.zlen}
    axis = min(dims, key=lambda k: dims[k])

    def edge_center(e):
        # For circular edges, arcCenter is more robust; fallback to Center()
        try:
            return e.arcCenter()
        except Exception:
            return e.Center()

    def proj_and_radial(p):
        if axis == "X":
            u = p.y - c0.y
            v = p.z - c0.z
        elif axis == "Y":
            u = p.x - c0.x
            v = p.z - c0.z
        else:  # "Z"
            u = p.x - c0.x
            v = p.y - c0.y
        return (u, v, math.hypot(u, v))

    # Gather small circular edges and pick the one farthest from center (likely the valve hole)
    circ = []
    for e in shape.Edges():
        try:
            if e.geomType() != "CIRCLE":
                continue
            r = float(e.radius())
            if not (1.0 <= r <= 15.0):
                continue
            p = edge_center(e)
            u, v, rr = proj_and_radial(p)
            circ.append((rr, r, u, v, e))
        except Exception:
            continue

    print(f"Loaded: {step_path}")
    print(f"BBox lens (X,Y,Z)=({bbox.xlen:.3f},{bbox.ylen:.3f},{bbox.zlen:.3f}); inferred axis={axis}")
    print(f"Small circular edges found: {len(circ)}")

    if not circ:
        print("No candidate circular edges found for valve hole; returning original model")
        return model

    circ.sort(key=lambda t: t[0], reverse=True)
    rr0, r0, u0, v0, _ = circ[0]
    print(f"Valve candidate: radius={r0:.4f}mm, radial_dist={rr0:.3f}mm (proj u,v=({u0:.3f},{v0:.3f}))")

    tol_r = 0.10  # mm
    tol_uv = 0.30  # mm

    def is_valve_edge(e):
        try:
            if e.geomType() != "CIRCLE":
                return False
            r = float(e.radius())
            if abs(r - r0) > tol_r:
                return False
            p = edge_center(e)
            u, v, _rr = proj_and_radial(p)
            if math.hypot(u - u0, v - v0) > tol_uv:
                return False
            return True
        except Exception:
            return False

    selected = model.edges().filter(is_valve_edge)
    try:
        nsel = selected.size()
    except Exception:
        # size() might not exist in older CQ; best-effort count
        try:
            nsel = len(selected.vals())
        except Exception:
            nsel = -1
    print(f"Valve-hole circular edges selected for chamfer: {nsel}")

    # Apply 1mm chamfer on both sides (both circular edges of the through-hole)
    result = selected.chamfer(1.0)
    return result
