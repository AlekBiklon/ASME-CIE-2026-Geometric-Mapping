def my_cad_function(args):
    import os
    import cadquery as cq

    # Load STEP from provided input_file
    if "input_file" not in args:
        raise ValueError("No input_file argument provided")

    step_path = os.path.expanduser(args["input_file"])
    model = cq.importers.importStep(step_path)

    # Debug info on first iteration
    occ = model.val() if hasattr(model, "val") else model
    try:
        print(f"Loaded STEP: {step_path}")
        print(f"Is Valid: {occ.isValid()}")
        try:
            solids = occ.Solids()
            print(f"Solids: {len(solids)}")
        except Exception:
            pass
        print(f"Faces: {len(occ.Faces())}")
        print(f"Edges: {len(occ.Edges())}")
        bb = occ.BoundingBox()
        c = bb.center
        print(f"BBox: x=[{bb.xmin:.3f},{bb.xmax:.3f}] y=[{bb.ymin:.3f},{bb.ymax:.3f}] z=[{bb.zmin:.3f},{bb.zmax:.3f}]")
        print(f"BBox Center: ({c.x:.3f}, {c.y:.3f}, {c.z:.3f})")
    except Exception as e:
        print(f"Debug info failed: {e}")

    # Add small fillet (0.2 mm) along sharp edges (apply broadly to all edges)
    try:
        result = model.edges().fillet(0.2)
        print("Applied 0.2 mm fillet to all edges.")
        return result
    except Exception as e:
        print(f"Global fillet(0.2) failed: {e}")

    # Fallback attempt on solids selection
    try:
        result = model.solids().edges().fillet(0.2)
        print("Applied 0.2 mm fillet to all solid edges (fallback).")
        return result
    except Exception as e:
        print(f"Fallback fillet(0.2) on solids failed: {e}")

    # If filleting fails, return original model
    print("Returning original model (no fillet applied due to errors).")
    return model
