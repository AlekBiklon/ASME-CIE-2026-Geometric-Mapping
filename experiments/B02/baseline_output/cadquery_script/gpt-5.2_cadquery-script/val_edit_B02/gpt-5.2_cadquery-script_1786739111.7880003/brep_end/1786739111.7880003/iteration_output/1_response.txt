    radius = 0.2  # mm

    def _fillet_solid_edges(solid, min_len):
        """Fillet edges on a solid, filtering tiny edges to improve robustness."""
        edges = list(solid.Edges())
        if min_len is not None:
            edges = [e for e in edges if e.Length() > float(min_len)]
        if not edges:
            return solid
        return solid.fillet(radius, edges)

    # --- Preferred approach: fillet each solid's edges (more robust than Workplane selection)
    try:
        solids = list(shape.Solids())
        if solids:
            filleted = []
            # progressively more selective if needed
            for s in solids:
                try:
                    filleted.append(_fillet_solid_edges(s, min_len=0.0))

                        filleted.append(_fillet_solid_edges(s, min_len=0.5))

            if len(filleted) == 1:

                return cq.Workplane(obj=comp)
    except Exception as e:
        print(f"Per-solid fillet attempt failed: {e}")

    # --- Fallback: try CadQuery Workplane edge selection (may fail on complex geometry)
    try:

    print("Returning original model (no fillet applied due to errors).")
    return model
