    def _is_step(p):

    def _collect_step_files(search_roots, max_files=200):
        found = []
        for root in search_roots:
            if not root or not isinstance(root, str):
                continue
            root = os.path.expanduser(root)
            if not os.path.exists(root):
                continue
            if os.path.isfile(root) and _is_step(root):
                found.append(os.path.abspath(root))
                continue
            if os.path.isdir(root):
                for dirpath, dirnames, filenames in os.walk(root):
                    for fn in filenames:
                        if fn.lower().endswith((".step", ".stp")):
                            found.append(os.path.abspath(os.path.join(dirpath, fn)))
                            if len(found) >= max_files:
                                return found
        return found

                pass

        return [p for _, p in ranked]
