def my_cad_function(args):
    import os
    import cadquery as cq

    # Load the provided STEP model
    if "input_file" not in args:
        raise ValueError("No input_file provided")

    step_path = os.path.expanduser(args["input_file"])
    model = cq.importers.importStep(step_path)

    # Get underlying shape for debug
    shape = model.val() if hasattr(model, "val") else model
    try:
        bbox = shape.BoundingBox()
        c = bbox.center
        print(f"Loaded STEP: {os.path.basename(step_path)}")
        print(f"Valid: {shape.isValid()}")
        print(f"Faces: {len(shape.Faces())}, Edges: {len(shape.Edges())}")
        print(f"Bbox: x={bbox.xlen:.3f}, y={bbox.ylen:.3f}, z={bbox.zlen:.3f}  center=({c.x:.3f},{c.y:.3f},{c.z:.3f})")
    except Exception as e:
        print(f"Debug info failed: {e}")

    # Apply a small safety fillet to sharp edges
    r = 0.2  # mm
    try:
        result = cq.Workplane(obj=shape).edges().fillet(r)
        print(f"Applied fillet r={r} mm to all edges.")
        return result
    except Exception as e:
        # Fallback: try filleting only straight (LINE) edges if full-edge fillet fails
        print(f"Fillet-all-edges failed: {e}")
        try:
            result = cq.Workplane(obj=shape).edges("%LINE").fillet(r)
            print(f"Applied fillet r={r} mm to LINE edges only (fallback).")
            return result
        except Exception as e2:
            print(f"Fallback fillet also failed: {e2}")
            # Return original if filleting fails
            return model
