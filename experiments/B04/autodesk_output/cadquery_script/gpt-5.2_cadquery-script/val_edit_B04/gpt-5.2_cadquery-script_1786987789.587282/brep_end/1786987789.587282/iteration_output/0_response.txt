    # Load the provided STEP model
    step_path = None

    # Add a 2 mm diameter hole through the nipple along the vertical axis (+/-Y)
    hole_d = 2.0
    hole_r = hole_d / 2.0

    # Assume nipple is centered near bbox center in XZ
    cx, cz = bb.center.x, bb.center.z
