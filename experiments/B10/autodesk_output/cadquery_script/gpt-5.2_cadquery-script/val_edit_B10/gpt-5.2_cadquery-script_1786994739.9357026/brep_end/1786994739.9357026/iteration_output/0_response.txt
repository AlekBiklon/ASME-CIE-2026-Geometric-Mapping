def my_cad_function(args):
    import os
    import cadquery as cq

    step_path = None
    if "input_file" in args and args["input_file"]:
        step_path = os.path.expanduser(args["input_file"])

    if not step_path or not os.path.exists(step_path):
        print(f"STEP input_file not found: {step_path}")
        return None

    model = cq.importers.importStep(step_path)
    solid = model.val() if hasattr(model, "val") else model

    # Debug info (useful on first iteration)
    try:
        bb = solid.BoundingBox()
        c = bb.center
        print(f"Loaded STEP: {os.path.basename(step_path)}")
        print(f"Valid: {solid.isValid()}")
        print(f"Faces: {len(solid.Faces())}, Edges: {len(solid.Edges())}")
        print(
            f"BBox: x[{bb.xmin:.3f},{bb.xmax:.3f}] y[{bb.ymin:.3f},{bb.ymax:.3f}] z[{bb.zmin:.3f},{bb.zmax:.3f}] center=({c.x:.3f},{c.y:.3f},{c.z:.3f})"
        )
    except Exception as e:
        print(f"Debug bbox failed: {e}")

    # Collect ONLY outer-boundary edges (outer wires) of all faces.
    # This generally excludes hole-mouth edges (which are inner wires) and targets the visible/external edges.
    edge_map = {}
    for f in solid.Faces():
        try:
            ow = f.outerWire()
            for e in ow.Edges():
                try:
                    key = e.hashCode()
                except Exception:
                    # Fallback: hash wrapped shape
                    key = e.wrapped.HashCode(1000003)
                edge_map[key] = e
        except Exception:
            continue

    outer_edges = list(edge_map.values())
    print(f"Outer-wire unique edges selected for fillet: {len(outer_edges)}")

    wp = cq.Workplane("XY").add(solid)

    # Apply fillet radius 5 mm to selected (visible/external) edges
    try:
        result = wp.newObject(outer_edges).fillet(5.0)
        return result
    except Exception as e:
        print(f"Fillet(5.0) failed: {e}")
        import traceback
        traceback.print_exc()
        # Return original if fillet fails
        return wp
