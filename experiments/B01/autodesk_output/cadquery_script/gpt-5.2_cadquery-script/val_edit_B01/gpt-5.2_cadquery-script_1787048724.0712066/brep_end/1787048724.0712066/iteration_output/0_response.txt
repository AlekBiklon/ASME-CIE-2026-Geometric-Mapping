def my_cad_function(args):
    import os
    import cadquery as cq

    # Load the provided STEP model
    if "input_file" not in args:
        print("No input_file provided")
        return None

    step_path = os.path.expanduser(args["input_file"])
    model_wp = cq.importers.importStep(step_path)
    shape = model_wp.val() if hasattr(model_wp, "val") else model_wp

    # Target face info from the request event (Fusion selection)
    target_pt = cq.Vector(-0.9689042506737217, -0.040016009795494537, 1.75)
    target_z = 1.75

    # Debug info
    try:
        bb = shape.BoundingBox()
        c = bb.center
        print(f"Loaded STEP: {step_path}")
        print(f"Model bbox: x[{bb.xmin:.3f},{bb.xmax:.3f}] y[{bb.ymin:.3f},{bb.ymax:.3f}] z[{bb.zmin:.3f},{bb.zmax:.3f}]")
        print(f"Model bbox center: ({c.x:.3f}, {c.y:.3f}, {c.z:.3f})")
        print(f"Num faces: {len(shape.Faces())}")
        print(f"Target point: ({target_pt.x:.4f}, {target_pt.y:.4f}, {target_pt.z:.4f})")
    except Exception as e:
        print(f"Debug bbox failed: {e}")

    faces = shape.Faces()

    def dist2_xy(v1, v2):
        dx = v1.x - v2.x
        dy = v1.y - v2.y
        return dx*dx + dy*dy

    # Prefer a face whose center z is near 1.75 (planar Z face), otherwise fall back to nearest face.
    candidates = []
    for f in faces:
        try:
            fc = f.Center()
            if abs(fc.z - target_z) < 0.05:
                candidates.append((dist2_xy(fc, target_pt), f, fc))
        except Exception:
            pass

    if candidates:
        candidates.sort(key=lambda t: t[0])
        chosen_face = candidates[0][1]
        chosen_center = candidates[0][2]
        print(f"Chosen face (z-filtered) center: ({chosen_center.x:.4f}, {chosen_center.y:.4f}, {chosen_center.z:.4f})")
    else:
        # Fallback: nearest face to target point
        sel = cq.selectors.NearestToPointSelector(target_pt)
        chosen_wp = cq.Workplane(obj=shape).faces(sel)
        chosen_face = chosen_wp.val()
        try:
            chosen_center = chosen_face.Center()
            print(f"Chosen face (nearest fallback) center: ({chosen_center.x:.4f}, {chosen_center.y:.4f}, {chosen_center.z:.4f})")
        except Exception:
            print("Chosen face (nearest fallback), center unavailable")

    # Cut a 2mm diameter through-hole normal to the selected face
    result = (
        cq.Workplane(obj=shape)
        .faces(cq.selectors.NearestToPointSelector(chosen_face.Center() if hasattr(chosen_face, 'Center') else target_pt))
        .workplane(centerOption="CenterOfMass")
        .hole(2.0)
    )

    print("Added 2.0mm through-hole for reset button.")
    return result
