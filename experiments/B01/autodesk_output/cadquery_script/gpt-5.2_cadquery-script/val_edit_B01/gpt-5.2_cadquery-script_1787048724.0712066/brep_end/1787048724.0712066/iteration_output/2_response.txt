    if not os.path.exists(step_path):
        print("STEP path does not exist:", step_path)
        return None

    model_wp = cq.importers.importStep(step_path)
    shape = model_wp.val() if hasattr(model_wp, "val") else model_wp

    # Target point from Fusion selection event (face centroid)
    target_pt = cq.Vector(-0.9689042506737217, -0.040016009795494537, 1.75)

    # --- debug info ---
    try:
        bb = shape.BoundingBox()
        c = bb.center
        print(
            f"BBox: x[{bb.xmin:.3f},{bb.xmax:.3f}] y[{bb.ymin:.3f},{bb.ymax:.3f}] z[{bb.zmin:.3f},{bb.zmax:.3f}] center=({c.x:.3f},{c.y:.3f},{c.z:.3f})"
        )
        print(f"Faces: {len(shape.Faces())}")
        print(f"Target point: ({target_pt.x:.4f}, {target_pt.y:.4f}, {target_pt.z:.4f})")
    except Exception as e:
        print("Debug bbox failed:", e)

    # --- select face nearest to the indicated point ---
    sel = cq.selectors.NearestToPointSelector(target_pt)
    face_wp = cq.Workplane(obj=shape).faces(sel)

    try:
        chosen_face = face_wp.val()
        fc = chosen_face.Center()
        try:
            fa = chosen_face.Area()
        except Exception:
            fa = None
        print(
            "Chosen face center:",
            f"({fc.x:.4f},{fc.y:.4f},{fc.z:.4f})",
            "area:" if fa is not None else "",
            f"{fa:.6f}" if fa is not None else ""
        )
    except Exception as e:
        print("Failed to evaluate chosen face:", e)

    # Create a workplane on that face, then place a 2mm through-hole at the target location
    wp_on_face = face_wp.workplane(centerOption="CenterOfMass")

    # Convert global target point into local coordinates of this face workplane
    try:
        pl = wp_on_face.plane
        local = pl.toLocalCoords(target_pt)
        px, py = float(local.x), float(local.y)
        print(f"Local (face workplane) point: ({px:.4f}, {py:.4f})")
    except Exception as e:
        print("Local coord transform failed; using face origin. Error:", e)
        px, py = 0.0, 0.0

    # Cut the reset-button access hole (2mm diameter), normal to the selected face
    result = wp_on_face.center(px, py).hole(2.0)
