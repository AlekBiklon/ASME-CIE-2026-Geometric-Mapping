def my_cad_function(args):
    import os
    import cadquery as cq

    # --- locate input STEP ---
    step_path = None
    for k in ["input_file", "brep_start_path_step", "step_file", "step_path"]:
        if k in args and args[k]:
            step_path = os.path.expanduser(str(args[k]))
            break

    if not step_path:
        print("No STEP path found in args. Available keys:", sorted(list(args.keys())))
        return None

    print(f"Loading STEP: {step_path}")
    model_wp = cq.importers.importStep(step_path)
    shape = model_wp.val() if hasattr(model_wp, "val") else model_wp

    # Target point from Fusion selection event (face centroid)
    target_pt = cq.Vector(-0.9689042506737217, -0.040016009795494537, 1.75)

    # --- debug info ---
    try:
        bb = shape.BoundingBox()
        c = bb.center
        print(
            f"BBox: x[{bb.xmin:.3f},{bb.xmax:.3f}] y[{bb.ymin:.3f},{bb.ymax:.3f}] z[{bb.zmin:.3f},{bb.zmax:.3f}]  center=({c.x:.3f},{c.y:.3f},{c.z:.3f})"
        )
        print(f"Faces: {len(shape.Faces())}")
        print(f"Target point: ({target_pt.x:.4f}, {target_pt.y:.4f}, {target_pt.z:.4f})")
    except Exception as e:
        print("Debug bbox failed:", e)

    # --- pick the face nearest to the target point and cut a 2mm hole at that point ---
    sel = cq.selectors.NearestToPointSelector(target_pt)

    # Create a workplane on the chosen face with origin at the face COM
    wp_on_face = cq.Workplane(obj=shape).faces(sel).workplane(centerOption="CenterOfMass")

    # Convert the global target point into that workplane's local coordinates
    try:
        pl = wp_on_face.plane
        local = pl.toLocalCoords(target_pt)
        px, py = float(local.x), float(local.y)
        print(f"Local (workplane) point: ({px:.4f}, {py:.4f})")
    except Exception as e:
        # Fallback: if transform fails, place hole at face origin
        print("Plane local-coord transform failed; falling back to face origin. Error:", e)
        px, py = 0.0, 0.0

    # Through-hole for reset button
    result = wp_on_face.center(px, py).hole(2.0)

    print("Added 2.0mm through-hole for reset button.")
    return result
